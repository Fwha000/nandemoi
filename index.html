import pygame import sys import random # --- 設定 --- WIDTH, HEIGHT = 640, 480 FPS = 60 WHITE = (255, 255, 255) BLACK = (0, 0, 0) RED = (255, 0, 0) YELLOW = (255, 255, 0) pygame.init() screen = pygame.display.set_mode((WIDTH, HEIGHT)) pygame.display.set_caption("Undertale-like Engine") clock = pygame.time.Clock() font = pygame.font.SysFont("arial", 24) # --- クラス定義 --- class Heart: """プレイヤーが操作する魂（赤）""" def __init__(self): self.rect = pygame.Rect(WIDTH//2, HEIGHT//2 + 50, 16, 16) self.speed = 4 def update(self, boundary): keys = pygame.key.get_pressed() if keys[pygame.K_LEFT] and self.rect.left > boundary.left: self.rect.x -= self.speed if keys[pygame.K_RIGHT] and self.rect.right < boundary.right: self.rect.x += self.speed if keys[pygame.K_UP] and self.rect.top > boundary.top: self.rect.y -= self.speed if keys[pygame.K_DOWN] and self.rect.bottom < boundary.bottom: self.rect.y += self.speed def draw(self): pygame.draw.rect(screen, RED, self.rect) class Bullet: """敵が放つ弾幕""" def __init__(self, boundary): self.rect = pygame.Rect(random.randint(boundary.left, boundary.right), boundary.top - 20, 10, 10) self.speed = random.randint(3, 6) def update(self): self.rect.y += self.speed def draw(self): pygame.draw.ellipse(screen, WHITE, self.rect) # --- メインループ --- def main(): heart = Heart() bullets = [] # バトルボックス（枠）の定義 box_rect = pygame.Rect(WIDTH//2 - 100, HEIGHT//2, 200, 150) dialogue_text = "* A mysterious AI blocks your way." hp = 20 frame_count = 0 while True: screen.fill(BLACK) for event in pygame.event.get(): if event.type == pygame.QUIT: pygame.quit() sys.exit() # 1. 弾幕の生成（時間経過で追加） frame_count += 1 if frame_count % 20 == 0: bullets.append(Bullet(box_rect)) # 2. アップデート heart.update(box_rect) for b in bullets[:]: b.update() # 当たり判定 if heart.rect.colliderect(b.rect): hp -= 1 bullets.remove(b) # 画面外で削除 elif b.rect.top > box_rect.bottom: bullets.remove(b) # 3. 描画 # モンスター（簡易的な形） pygame.draw.polygon(screen, WHITE, [(320, 50), (280, 150), (360, 150)]) # テキスト表示 text_surface = font.render(dialogue_text, True, WHITE) screen.blit(text_surface, (50, 200)) # HP表示 hp_text = font.render(f"HP: {hp} / 20", True, WHITE) screen.blit(hp_text, (WIDTH//2 - 50, HEIGHT - 50)) # バトルボックス pygame.draw.rect(screen, WHITE, box_rect, 3) # プレイヤーと弾 heart.draw() for b in bullets: b.draw() if hp <= 0: print("GAME OVER") break pygame.display.flip() clock.tick(FPS) if __name__ == "__main__": main()
